Security & policy

In this exercise the main focus is secure image signing with Cosign. On top of that, the design is ready to plug in OPA policies (for example in an admission controller or as a dedicated policy stage in the pipeline) to enforce additional rules on what is allowed to run in the cluster.

Image integrity

Cosign + ECR

Every image pushed to ECR is signed with Cosign. Before each deploy, the pipeline uses the public key to verify that the digest matches the original artifact.

cosign sign $(IMAGE) cosign verify $(IMAGE) immutable tags digest-based deploys

OPA policy options

OPA + Renovate

While Cosign is the core of this challenge, OPA could be used to add extra guardrails either in-cluster (admission controller) or as a dedicated policy stage in the pipeline. Typical policies for this setup could be:

  • Only allow containers whose images are signed by the expected Cosign key.
  • Enforce allowed registries (e.g. only this AWS ECR account).
  • Block deployments if the last security scan still has CRITICAL vulnerabilities.
  • Restrict which teams can deploy into prod namespace.
  • Ensure labels/annotations for traceability and auditability are present.

# Stage 0 – Terraform (EKS + ECR + IAM)
- stage: ProvisionInfra
  displayName: "Provision AWS Infrastructure (EKS + ECR + IAM)"
  condition: ne(variables['Build.Reason'], 'Schedule')
  jobs:
    - job: terraform_apply
      pool:
        vmImage: 'ubuntu-latest'
      steps:
        - checkout: self

        - task: TerraformInstaller@1
          inputs:
            terraformVersion: '1.13.5'

        - script: |
            set -euo pipefail
            cd infra
            terraform init
            terraform plan -var-file="envs/dev/terraform.tfvars" -out=tfplan
            terraform show -no-color tfplan
            terraform apply tfplan
          displayName: "Terraform plan + apply (dev only)"
          env:
            AWS_ACCESS_KEY_ID: $(AWS_ACCESS_KEY_ID)
            AWS_SECRET_ACCESS_KEY: $(AWS_SECRET_ACCESS_KEY)
            AWS_REGION: $(AWS_REGION)

        - script: |
            cd infra
            terraform output -json > tf_outputs.json
            echo "##vso[task.setvariable variable=ECR_REPO_URL;isOutput=true]$(jq -r .ecr_repository_url.value tf_outputs.json)"
            echo "##vso[task.setvariable variable=EKS_CLUSTER_NAME_DEV;isOutput=true]$(jq -r .eks_cluster_name.value tf_outputs.json)"
          name: setTerraformOutputs
          displayName: "Export Terraform outputs"

# Stage 1 – Renovate
- stage: Renovate
  displayName: "Renovate dependencies"
  condition: or(eq(variables['Build.Reason'], 'Schedule'), eq(variables['RUN_RENOVATE'], 'true'))
  jobs:
    - job: renovate
      pool:
        vmImage: 'ubuntu-latest'
      steps:
        - checkout: self
        - bash: |
            set -euo pipefail
            docker run --rm \
              -e RENOVATE_TOKEN=$(RENOVATE_TOKEN) \
              -e RENOVATE_PLATFORM=azure \
              -e RENOVATE_ENDPOINT=https://dev.azure.com \
              -e LOG_LEVEL=info \
              ghcr.io/renovatebot/renovate:latest \
              "$(System.CollectionUri)$(System.TeamProject)/_git/$(Build.Repository.Name)"
          displayName: "Run Renovate"
          env:
            RENOVATE_TOKEN: $(RENOVATE_TOKEN)

# Stage 2 – Build, test & Trivy (CRITICAL only)
- stage: BuildTestScan
  displayName: "Build, test & security scan"
  condition: ne(variables['Build.Reason'], 'Schedule')
  jobs:
    - job: build_test_scan
      pool:
        vmImage: 'ubuntu-latest'
      steps:
        - checkout: self
        - bash: |
            set -euo pipefail
            make test
          displayName: "Maven tests"
        - bash: |
            set -euo pipefail
            docker build -f app/Dockerfile -t "$(IMAGE_FULL)" app
          displayName: "Build Docker image"
        - bash: |
            set -euo pipefail
            if ! command -v trivy >/dev/null 2>&1; then
              curl -sfL https://raw.githubusercontent.com/aquasecurity/trivy/main/contrib/install.sh \
                | sh -s -- -b /usr/local/bin
            fi
            trivy image --severity CRITICAL --exit-code 1 --no-progress "$(IMAGE_FULL)"
          displayName: "Trivy scan (CRITICAL only)"
        - publish: $(Build.SourcesDirectory)
          artifact: src

# Stage 3 – Push to ECR & Cosign
- stage: PushAndSign
  displayName: "Push to ECR & Cosign"
  dependsOn: BuildTestScan
  condition: succeeded()
  jobs:
    - job: push_sign
      pool:
        vmImage: 'ubuntu-latest'
      steps:
        - download: current
          artifact: src
        - bash: |
            set -euo pipefail
            aws ecr describe-repositories --repository-names "$(ECR_REPO)" --region "$(AWS_REGION)" >/dev/null 2>&1 \
              || aws ecr create-repository --repository-name "$(ECR_REPO)" --region "$(AWS_REGION)" >/dev/null
            aws ecr get-login-password --region "$(AWS_REGION)" \
              | docker login --username AWS --password-stdin "$(CONTAINER_REGISTRY)"
            docker images | grep -q "$(IMAGE_FULL)" || docker build -f app/Dockerfile -t "$(IMAGE_FULL)" app
            docker push "$(IMAGE_FULL)"
            echo "$(IMAGE_FULL)" > image.txt
          displayName: "Push image to ECR"
          env:
            AWS_ACCESS_KEY_ID: $(AWS_ACCESS_KEY_ID)
            AWS_SECRET_ACCESS_KEY: $(AWS_SECRET_ACCESS_KEY)
            AWS_REGION: $(AWS_REGION)
        - bash: |
            set -euo pipefail
            curl -sSL https://github.com/sigstore/cosign/releases/latest/download/cosign-linux-amd64 \
              -o cosign && chmod +x cosign && sudo mv cosign /usr/local/bin/
            echo "$(COSIGN_PRIVATE_KEY_B64)" | base64 -d > cosign.key
            echo "$(COSIGN_PUBLIC_KEY_B64)"  | base64 -d > cosign.pub
            COSIGN_PASSWORD="${COSIGN_PASSWORD:-}" cosign sign --key cosign.key "$(IMAGE_FULL)"
          displayName: "Cosign sign image"
        - bash: |
            set -euo pipefail
            curl -sSL https://github.com/sigstore/cosign/releases/latest/download/cosign-linux-amd64 \
              -o cosign && chmod +x cosign && sudo mv cosign /usr/local/bin/
            echo "$(COSIGN_PUBLIC_KEY_B64)" | base64 -d > cosign.pub
            cosign verify --key cosign.pub "$(IMAGE_FULL)"
          displayName: "Cosign verify (gate)"
        - task: PublishBuildArtifacts@1
          inputs:
            pathToPublish: image.txt
            artifactName: image-ref

# Stage 4 – Deploy to dev (namespace dev)
- stage: Deploy_Dev
  displayName: "Deploy to dev"
  dependsOn: PushAndSign
  condition: succeeded()
  jobs:
    - deployment: deploy_dev
      environment: "dev"
      strategy:
        runOnce:
          deploy:
            steps:
              - download: current
                artifact: image-ref
              - bash: echo "IMAGE=$(cat $(Pipeline.Workspace)/image-ref/image.txt)" >> $(Pipeline.Workspace)/image.env
              - bash: |
                  set -euo pipefail
                  source $(Pipeline.Workspace)/image.env
                  curl -sSL https://github.com/sigstore/cosign/releases/latest/download/cosign-linux-amd64 \
                    -o cosign && chmod +x cosign && sudo mv cosign /usr/local/bin/
                  echo "$(COSIGN_PUBLIC_KEY_B64)" | base64 -d > cosign.pub
                  cosign verify --key cosign.pub "${IMAGE}"
                displayName: "Verify signature (dev gate)"
              - bash: |
                  set -euo pipefail
                  source $(Pipeline.Workspace)/image.env
                  aws eks update-kubeconfig --region "$(AWS_REGION)" --name "$(CLUSTER_NAME_DEV)"
                  command -v helm >/dev/null || (curl https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash)
                  ./scripts/deploy.sh dev "${IMAGE}" ./helm/nn-devops-challenge
                displayName: "Helm deploy to dev"
                env:
                  AWS_ACCESS_KEY_ID: $(AWS_ACCESS_KEY_ID)
                  AWS_SECRET_ACCESS_KEY: $(AWS_SECRET_ACCESS_KEY)
                  AWS_REGION: $(AWS_REGION)
